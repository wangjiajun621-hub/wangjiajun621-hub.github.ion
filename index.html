Add this code to your Colab notebook after training to export the model:

import torch
# After training completes, export best model to ONNX
model = build_model(final_cfg['model_name'], drop=final_cfg['dropout'])
unfreeze_backbone(model, final_cfg['model_name'])
model.load_state_dict({k: v.to(DEVICE) for k,v in final_result['best_state'].items()})
model.eval()

img_size = final_cfg['img_size']
dummy = torch.randn(1, 3, img_size, img_size).to(DEVICE)

torch.onnx.export(
    model, dummy,
    "bone_fracture.onnx",
    input_names=["input"],
    output_names=["output"],
    dynamic_axes={"input": {0: "batch"}, "output": {0: "batch"}},
    opset_version=17,
)
print("✅ Exported: bone_fracture.onnx")
# Download from Colab: files.download("bone_fracture.onnx")
